# Voidline Arch Linux dependency manifest
#
# install.sh reads this file. Format: a `[section]` line, then package names
# separated by spaces or newlines. `#` starts a comment. Every name here was
# checked against the Arch repositories and the AUR.
#
#   required     the shell does not work correctly without these
#   build        needed to build the Rust backend and network helper
#   recommended  installed with --install-deps; missing ones degrade the UI
#                (the icon font, image tools, file chooser portal, updates)
#   optional     feature integrations; installed with --with-optional
#   aur-recommended / aur-optional
#                only in the AUR; installed when paru or yay is available
#   sddm / lyra  installed with --with-sddm / --with-lyra

[required]
# Compositor, shell toolkit, and portals
hyprland quickshell xdg-desktop-portal-hyprland qt6-declarative qt6-svg
qt6-imageformats
# Session services the shell talks to
systemd polkit networkmanager upower power-profiles-daemon
pipewire pipewire-pulse wireplumber bluez bluez-utils
# Tools called by the shell and its scripts
brightnessctl playerctl jq curl wl-clipboard cliphist grim slurp libnotify
util-linux iproute2 ripgrep findutils file xdg-utils xdg-user-dirs glib2
# Voidline Terminal and Settings runtime libraries
gtk4 vte4

[build]
pkgconf gcc

[recommended]
# Material Symbols provides every icon in the shell; without it icons render
# as their text names.
ttf-material-symbols-variable noto-fonts papirus-icon-theme
xdg-desktop-portal-gtk imagemagick pacman-contrib pipewire-alsa hypridle
qrencode
# File manager bound to Super+E in the bundled Hyprland config
nautilus

[optional]
# Screen recording, colour picker, night light
wf-recorder hyprpicker hyprsunset
# External monitor brightness
ddcutil
# Hotspot and Wi-Fi tools
iw hostapd dnsmasq iptables
# Hardware information
lm_sensors usbutils pciutils udisks2 v4l-utils
# Printers and scanners
cups sane sane-airscan avahi
# Security and networking settings
ufw wireguard-tools
# Accessibility
orca
# Colour management
colord
# iwd credential helper
python python-dbus python-gobject

[aur-recommended]
# Voidline's interface typeface; Qt falls back to the default sans font.
ttf-roboto-flex

[aur-optional]
wvkbd linux-wifi-hotspot

[sddm]
sddm qt5-graphicaleffects qt5-quickcontrols2

[lyra]
ollama
